```python
"""pyinstrumentベースのプロファイリングユーティリティ。"""

from __future__ import annotations

import logging
from pathlib import Path
from typing import Any

logger = logging.getLogger(__name__)


class ProfileContext:
    """pyinstrumentプロファイリングコンテキストマネージャー。

    --profileフラグ指定時のみプロファイリングを実行。
    未指定時はno-op (オーバーヘッドなし)。

    Usage:
        with ProfileContext(enabled=args.profile, label='backtest'):
            run_backtest(...)
    """

    def __init__(self, enabled: bool = False, label: str = 'profile') -> None:
        self._enabled = enabled
        self._label = label
        self._profiler: Any = None  # pyinstrument.Profiler or None

    def __enter__(self) -> "ProfileContext":
        if self._enabled:
            try:
                from pyinstrument import Profiler

                self._profiler = Profiler()
                self._profiler.start()
            except ImportError:
                logger.warning(
                    "pyinstrument is not installed. "
                    "Install with: pip install pyinstrument"
                )
                self._enabled = False
        return self

    def __exit__(self, *args: Any) -> None:
        if self._profiler is not None:
            self._profiler.stop()
            # テキスト出力 → stdout
            text_output = self._profiler.output_text(unicode=True, color=True)
            print(text_output)
            # HTML出力 → ファイル
            output_dir = Path("data/profiles")
            output_dir.mkdir(parents=True, exist_ok=True)
            html_path = output_dir / f"{self._label}.html"
            self._profiler.write_html(str(html_path))
            logger.info("Profile saved: %s", html_path)
```

Key design decisions (Claude's discretion per CONTEXT.md):
- Context manager pattern (same as TimingContext)
- Lazy import of pyinstrument inside `__enter__` — avoids ImportError when not profiling
- Graceful degradation: if pyinstrument not installed, log warning and continue without profiling
- HTML saved to `data/profiles/{label}.html` per D-16
- Text output to stdout per D-16
- No overhead when `enabled=False` per D-17

**1. Add cache utility functions to feature_engine.py (module-level, after imports):**

```python
import hashlib
import json
from pathlib import Path


def compute_cache_key(
    input_paths: list[Path],
    date_range: tuple[str, str] | None,
    feature_type: str,
) -> str:
    """キャッシュキーを計算: 入力パス + 日付範囲 + 特徴量種別 → SHA-256先頭16文字"""
    payload = json.dumps(
        {
            "paths": [str(p) for p in sorted(input_paths)],
            "start": date_range[0] if date_range else "",
            "end": date_range[1] if date_range else "",
            "type": feature_type,
        },
        sort_keys=True,
    )
    return hashlib.sha256(payload.encode()).hexdigest()[:16]


def is_cache_valid(
    cache_path: Path,
    source_paths: list[Path],
) -> bool:
    """ハイブリッド無効化: タイムスタンプ比較(高速) → コンテンツハッシュ(厳密)

    Returns True if cache is valid (can be used), False if stale.
    """
    if not cache_path.exists():
        return False
    cache_mtime = cache_path.stat().st_mtime
    for src in source_paths:
        if not src.exists():
            return False
        if src.stat().st_mtime > cache_mtime:
            return False
    return True
```

        # Check both .parquet file and directory forms

            source_paths.append(p.with_suffix(".parquet"))
        elif p.is_dir():

            if not cached_df.empty:
                return cached_df
        except Exception:
