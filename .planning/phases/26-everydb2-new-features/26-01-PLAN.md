    FEATURE_COLS: list[str] = ["dm_time_rank", "dm_time_zscore", "dm_confidence_range", "dm_time_margin_to_fav"]

    class MiningFeatures:
        def __init__(self, store: ParquetStore) -> None:
            self.store = store
            self._mining_cache: pd.DataFrame | None = None

        def _load_mining(self) -> pd.DataFrame:
            """Load mining.parquet with caching. Returns empty DataFrame if not available."""
            if self._mining_cache is None:
                if self.store.exists("raw", "mining"):
                    self._mining_cache = self.store.read("raw", "mining")
                else:
                    self._mining_cache = pd.DataFrame()
            return self._mining_cache

    Implement _pivot_mining_to_long(mining_df: pd.DataFrame) -> pd.DataFrame:
        - For i in range(1, 19):
          - Rename Umaban{i} -> "umaban", DMTime{i} -> "dm_time", DMGosaP{i} -> "dm_gosa_plus", DMGosaM{i} -> "dm_gosa_minus"
          - Keep race_id column (computed from Year+MonthDay+JyoCD+Kaiji+Nichiji+RaceNum via _compute_race_id if not present, or use existing race_id)
          - Filter out rows where umaban is NaN or "sp" (empty slot initial value)
        - Concat all 18 subsets into long format
        - Convert dm_time, dm_gosa_plus, dm_gosa_minus to float (they are varchar in source)
        - Convert umaban to int
        Return the long-format DataFrame

    Implement compute(entry_df: pd.DataFrame) -> pd.DataFrame:
        1. Load mining data via _load_mining()
        2. If empty, return entry_df[["race_id", "umaban"]] with NaN for all FEATURE_COLS
        3. Filter to DataKubun="3" (final prediction). If no DataKubun=3, fallback to DataKubun="2"
        4. Call _pivot_mining_to_long() to convert wide to long
        5. Merge long mining with entry_df on (race_id, umaban)
        6. Compute features:
           - dm_time_rank: groupby("race_id").rank(method="min", ascending=True) on dm_time. Lower predicted time = rank 1.
           - dm_time_zscore: groupby("race_id") z-score: (dm_time - mean) / std, with std=0 fallback to 0.0
           - dm_confidence_range: dm_gosa_plus + dm_gosa_minus (total prediction uncertainty spread)
           - dm_time_margin_to_fav: dm_time - groupby("race_id").min() (gap to fastest predicted horse)
        7. Return result[["race_id", "umaban"] + FEATURE_COLS]

    NOTE on n_mining ETL format: The etl_tables.yaml defines pk with "umaban" but the actual table has Umaban1-18 wide format. The ETL may output the raw wide format to mining.parquet. The _pivot_mining_to_long function handles this. Check actual parquet schema after ETL to confirm format.

    Also implement compute_batch(df: pd.DataFrame) -> pd.DataFrame as an alternative for _train_submodel integration:
        - Takes df with race_id, umaban columns
        - Calls compute(df) internally
        - Returns same result format
